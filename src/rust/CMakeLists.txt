# Builds librustzcash.a (the Rust FFI, crate-type = staticlib) via cargo, then
# exposes it as an IMPORTED target. Mirrors the cargo-build rule and the
# RUST_ENV_VARS / offline .cargo/config generation in src/Makefile.am.

find_program(CARGO_BIN NAMES cargo HINTS ${DEPENDS_PREFIX}/native/bin REQUIRED)
find_program(RUSTC_BIN NAMES rustc HINTS ${DEPENDS_PREFIX}/native/bin REQUIRED)

set(_vendored_sources ${DEPENDS_PREFIX}/vendored-sources)
set(_rust_lib ${CMAKE_SOURCE_DIR}/target/${RUST_TARGET}/release/librustzcash.a)

# Reproduce depends/../.cargo -> offline config: persistent-config plus the
# vendored-sources replacement with the real directory substituted in.
file(READ ${CMAKE_SOURCE_DIR}/.cargo/persistent-config _persistent_cfg)
file(WRITE ${CMAKE_SOURCE_DIR}/.cargo/config.toml
"${_persistent_cfg}
# Generated by src/rust/CMakeLists.txt for the vendored offline build:
[source.crates-io]
replace-with = \"vendored-sources\"

[source.vendored-sources]
directory = \"${_vendored_sources}\"
")

file(GLOB_RECURSE _rust_srcs CONFIGURE_DEPENDS
  ${CMAKE_SOURCE_DIR}/src/rust/src/*.rs
)

add_custom_command(
  OUTPUT ${_rust_lib}
  COMMAND ${CMAKE_COMMAND} -E env
    RUSTC=${RUSTC_BIN}
    "RUSTFLAGS=--cfg=rust_secp_no_symbol_renaming -L native=${CMAKE_SOURCE_DIR}/src/secp256k1/.libs"
    CC=${CMAKE_C_COMPILER}
    "CFLAGS=${CMAKE_C_FLAGS}"
    CXX=${CMAKE_CXX_COMPILER}
    "CXXFLAGS=${CMAKE_CXX_FLAGS}"
    TERM=dumb
    ${CARGO_BIN} build --release --offline
      --target ${RUST_TARGET}
      --manifest-path ${CMAKE_SOURCE_DIR}/Cargo.toml
  DEPENDS ${CMAKE_SOURCE_DIR}/Cargo.toml ${CMAKE_SOURCE_DIR}/Cargo.lock ${_rust_srcs} secp256k1_ep
  COMMENT "cargo build librustzcash (${RUST_TARGET}, offline)"
  VERBATIM
  USES_TERMINAL
)
add_custom_target(rust_ffi DEPENDS ${_rust_lib})

add_library(librustzcash STATIC IMPORTED GLOBAL)
set_target_properties(librustzcash PROPERTIES IMPORTED_LOCATION ${_rust_lib})
add_dependencies(librustzcash rust_ffi)
